Program 3: Reverse of a Number Using UDF

#include <stdio.h>

void reverse()
{
    int n, rem, rev = 0;

    printf("Enter a number: ");
    scanf("%d", &n);

    while(n > 0)
    {
        rem = n % 10;
        rev = rev * 10 + rem;
        n = n / 10;
    }

    printf("Reverse = %d", rev);
}

int main()
{
    reverse();
    return 0;
}
